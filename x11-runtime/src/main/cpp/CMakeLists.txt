cmake_minimum_required(VERSION 3.22)
project(magicdesk_x11 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../../../../vendor/magicdesk-x11/lorie/src/main/cpp engine)
add_subdirectory(../../../../hosted-runtime/src/main/cpp/graphics graphics)
add_library(Xlorie SHARED x11_jni.cpp x11_graphics.cpp)
target_include_directories(Xlorie PRIVATE ../../../../hosted-runtime/src/main/cpp)
target_link_libraries(Xlorie PRIVATE LorieNative magicdesk_graphics android log)
target_compile_options(Xlorie PRIVATE -fno-exceptions -fno-rtti)
target_link_options(Xlorie PRIVATE -nostdlib++ -Wl,--no-undefined)
