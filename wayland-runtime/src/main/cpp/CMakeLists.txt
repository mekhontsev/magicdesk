function(mdw_protocol name xml)
	foreach(kind server-header client-header private-code)
		if(kind STREQUAL "private-code")
			set(generated "${name}-protocol.c")
		elseif(kind STREQUAL "server-header")
			set(generated "${name}-protocol.h")
		else()
			set(generated "${name}-client-protocol.h")
		endif()
		add_custom_command(OUTPUT ${generated}
			COMMAND ${WAYLAND_SCANNER} ${kind} ${xml} ${CMAKE_CURRENT_BINARY_DIR}/${generated}
			DEPENDS ${xml})
	endforeach()
endfunction()
