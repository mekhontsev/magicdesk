cmake_minimum_required(VERSION 3.22)
project(MagicDeskGraphics LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(magicdesk_graphics STATIC graphics.c software.c)
target_include_directories(magicdesk_graphics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(magicdesk_graphics PRIVATE -Wall -Wextra -Werror)
target_link_libraries(magicdesk_graphics PUBLIC m)
include(CheckCSourceCompiles)
check_c_source_compiles("#ifndef __ANDROID__\n#error Not Android\n#endif\nint main(void) { return 0; }" MDG_ANDROID_BUILD)
if(MDG_ANDROID_BUILD)
    find_program(MDG_GLSLC glslc HINTS "${ANDROID_NDK}/shader-tools/linux-x86_64" "${ANDROID_NDK}/shader-tools/windows-x86_64" "${ANDROID_NDK}/shader-tools/darwin-x86_64")
    if(MDG_GLSLC)
        set(shader_compiler ${MDG_GLSLC} --target-env=vulkan1.1)
    else()
        find_program(MDG_GLSLANG glslangValidator REQUIRED)
        set(shader_compiler ${MDG_GLSLANG} -V --target-env vulkan1.1)
    endif()
    foreach(stage vert frag)
        add_custom_command(OUTPUT quad.${stage}.spv
            COMMAND ${shader_compiler} ${CMAKE_CURRENT_SOURCE_DIR}/shaders/quad.${stage} -o quad.${stage}.spv
            DEPENDS shaders/quad.${stage})
    endforeach()
    add_custom_command(OUTPUT shaders.h
        COMMAND ${CMAKE_COMMAND} -DVERTEX=${CMAKE_CURRENT_BINARY_DIR}/quad.vert.spv
            -DFRAGMENT=${CMAKE_CURRENT_BINARY_DIR}/quad.frag.spv -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/shaders.h
            -P ${CMAKE_CURRENT_SOURCE_DIR}/embed-shaders.cmake
        DEPENDS quad.vert.spv quad.frag.spv embed-shaders.cmake)
    target_sources(magicdesk_graphics PRIVATE vulkan.c dmabuf.c surface.c ${CMAKE_CURRENT_BINARY_DIR}/shaders.h)
    target_include_directories(magicdesk_graphics PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_compile_definitions(magicdesk_graphics PRIVATE MDG_ANDROID=1)
    target_link_libraries(magicdesk_graphics PUBLIC android dl log)
else()
    target_sources(magicdesk_graphics PRIVATE vulkan_stub.c)
endif()
option(MDG_BUILD_TESTS "Build native graphics fixtures" OFF)
if(MDG_BUILD_TESTS)
    enable_testing()
    add_executable(graphics-test ../../../../tests/native/graphics.c)
    target_compile_options(graphics-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(graphics-test PRIVATE magicdesk_graphics)
    add_test(NAME graphics-software COMMAND graphics-test --software)
    add_test(NAME graphics-auto COMMAND graphics-test)
    set_tests_properties(graphics-software graphics-auto PROPERTIES TIMEOUT 30)
    if(MDG_ANDROID_BUILD)
        add_executable(graphics-dmabuf-test ../../../../tests/native/dmabuf.c ../../../../tests/native/dmabuf_producer.c)
        target_compile_options(graphics-dmabuf-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        target_link_libraries(graphics-dmabuf-test PRIVATE magicdesk_graphics)
        add_test(NAME graphics-dmabuf COMMAND graphics-dmabuf-test)
        set_tests_properties(graphics-dmabuf PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    endif()
endif()
