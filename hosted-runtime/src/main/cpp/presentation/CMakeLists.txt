cmake_minimum_required(VERSION 3.22)
project(MagicDeskGraphicsHost LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_subdirectory(../graphics graphics)
add_library(magicdesk_graphics_host SHARED presenter.c)
if(JNI_INCLUDE_DIR)
    target_include_directories(magicdesk_graphics_host PRIVATE ${JNI_INCLUDE_DIR} ${JNI_INCLUDE_DIR}/linux)
endif()
target_compile_options(magicdesk_graphics_host PRIVATE -Wall -Wextra -Werror)
target_link_options(magicdesk_graphics_host PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,defs -Wl,--exclude-libs,ALL)
target_link_libraries(magicdesk_graphics_host PRIVATE magicdesk_graphics android)
