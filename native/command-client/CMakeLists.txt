cmake_minimum_required(VERSION 3.24)
project(MagicDeskCommandClient LANGUAGES C)
foreach(program client bridge)
    if(program STREQUAL client)
        set(source main.c)
    else()
        set(source bridge.c)
    endif()
    add_executable(command_${program} ${source} wire.c)
    set_target_properties(command_${program} PROPERTIES OUTPUT_NAME libmagicdesk_command_${program}.so)
    target_compile_options(command_${program} PRIVATE -std=c17 -O2 -Wall -Wextra -Werror -fno-pie)
    target_link_options(command_${program} PRIVATE -static -Wl,-z,noexecstack -Wl,-z,max-page-size=16384)
endforeach()
